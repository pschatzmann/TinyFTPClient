116 bool readDataLine(FTPBasicAPI *api) {
117 String line;
118 unsigned long start = millis();
119 while (true) {
120 if (stream_ptr->available() > 0) {
121 int c = stream_ptr->read();
122 if (c == '\n') break;
123 if (c >= 0) line += (char)c;
124 start = millis();
125 } else if (!api->dataConnected()) {
126 if (line.length() == 0) return false;
127 break; // last line without line feed
128 } else if (millis() - start >= FTP_REPLY_TIMEOUT_MS) {
129 FTPLogger.writeLog(LOG_ERROR, "FTPFileIterator", "timeout");
130 if (line.length() == 0) return false;
131 break;
132 } else {
133 delay(1);
134 }
135 }
136 // For Windows compatibility, remove trailing \r
137 if (line.endsWith("\r")) line.remove(line.length() - 1);
138 buffer = line;
139 return true;
140 }
