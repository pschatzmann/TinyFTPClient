204 size_t readData(Client *in, uint8_t *buf, size_t len, int terminator) {
205 FTPBasicAPI *api = transfer.api();
206 size_t n = 0;
207 unsigned long start = millis();
208 while (n < len) {
209 int avail = in->available();
210 if (avail > 0) {
211 if (terminator < 0) {
212 size_t chunk = len - n < (size_t)avail ? len - n : (size_t)avail;
213 int got = in->read(buf + n, chunk);
214 if (got > 0) n += got;
215 } else {
216 int c = in->read();
217 if (c == terminator) break;
218 if (c >= 0) buf[n++] = (uint8_t)c;
219 }
220 start = millis();
221 } else if (!api->dataConnected() || millis() - start >= getTimeout()) {
222 break;
223 } else {
224 delay(1);
225 }
226 }
227 return n;
228 }
