200 CallbackFile openFile(const char *path, const char *mode) {
201 report(Open, path, mode);
202 if (mode[0] == 'r') {
203 if (listCallback) {
204 std::vector<CallbackFileInfo> entries;

228inline int CallbackFile::read(uint8_t *data, size_t len) {
