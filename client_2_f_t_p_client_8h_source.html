171 FTPSession<ClientType> *idleSession() override {
172 for (auto &session : sessions) {
173 if (session != nullptr && session->currentOperation() == NOP &&
174 session->connected()) {
175 session->setUseTypeCommand(use_type_command);
176 return session;
177 }
178 }
179 // (re)connect an unused session: idle sessions are not connected here
180 for (auto &session : sessions) {
181 if (session == nullptr) {
182 session = new FTPSession<ClientType>();
183 } else if (session->currentOperation() != NOP) {
184 continue;
185 }
186 session->end();
187 if (!session->begin(remote_addr, port, userid, password)) {
188 FTPLogger.writeLog(LOG_ERROR, "FTPClient", "Connect failed");
189 session->end();
190 return nullptr;
191 }
192 session->setUseTypeCommand(use_type_command);
193 return session;
194 }
195 FTPLogger.writeLog(LOG_ERROR, "FTPClient", "No available sessions");
196 return nullptr;
197 }

201 FTPSession<ClientType> *openDataSession() override {
