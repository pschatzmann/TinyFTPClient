129 bool abort() {
130 bool rc = true;
131 if (current_operation == RESERVED_OP) {
132 // no transfer was started: just release the data connection
133 endTransfer("abort");
134 } else if (current_operation == READ_OP || current_operation == WRITE_OP ||
135 current_operation == LS_OP) {
136 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI", "abort");
137 closeData();
138 setCurrentOperation(NOP);
139
140 const char *ok[] = {"426", "226", "225", nullptr};
141 rc = cmd("ABOR", nullptr, ok);
142 if (rc && strncmp(result_reply, "426", 3) == 0) {
143 // 426 (transfer aborted) is followed by the reply to the ABOR
144 const char *ok_end[] = {"226", "225", nullptr};
145 rc = checkResult(ok_end, "abort-end");
146 } else {
147 // the transfer might have completed before the ABOR: then the reply
148 // to the ABOR is still coming
149 skipReplies(FTP_ABORT_DELAY_MS);
150 }
151 }
152 return rc;
153 }

227 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI::endTransfer", tag);
228 if (current_operation == READ_OP || current_operation == LS_OP) {
229 closeData();

285 bool checkResult(const char *expected[], const char *command,
286 bool wait_for_data = true) {
287 result_reply[0] = '\0';
288 // nothing to consume and we do not need to wait
289 if (!wait_for_data && command_ptr->available() == 0) return true;
290
291 char result_str[FTP_RESULT_BUFFER_SIZE];
292 bool ok = readReply(result_str, sizeof(result_str));
293 if (!ok) {
294 // if we got no reply and we dont need to wait we are still ok
295 ok = !wait_for_data;
296 } else {
297 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI::checkResult", result_str);
298 strncpy(result_reply, result_str, sizeof(result_reply) - 1);
299 result_reply[sizeof(result_reply) - 1] = '\0';
300 ok = isExpected(result_str, expected);
301 }
302
303 // log error
304 if (!ok) {
305 FTPLogger.writeLog(LOG_ERROR, "FTPBasicAPI::checkResult", command);
306 FTPLogger.writeLog(LOG_ERROR, "FTPBasicAPI::checkResult", result_reply);
307 }
308 return ok;
309 }

368 bool connect(IPAddress adr, int port, Client *client_ptr,
369 bool doCheckResult = false) {
370 char buffer[80];
371 snprintf(buffer, sizeof(buffer), "connect %d.%d.%d.%d:%d", adr[0], adr[1],
372 adr[2], adr[3], port);
373 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI::connect", buffer);
374 // make sure we start with a clean state
375 if (client_ptr->connected()) client_ptr->stop();
376 for (int j = 0; j < FTP_CONNECT_RETRIES; j++) {
377 if (j > 0) delay(FTP_CONNECT_RETRY_DELAY_MS);
378 if (client_ptr->connect(adr, port)) break;
379 }
380 bool ok = client_ptr->connected();
381 if (ok && doCheckResult) {
382 const char *ok_result[] = {"220", "200", nullptr};
383 ok = checkResult(ok_result, "connect");
384 }
385 // log result
386 if (ok) {
387 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI::connected", buffer);
388 } else {
389 FTPLogger.writeLog(LOG_ERROR, "FTPBasicAPI::connected", buffer);
390 }
391 return ok;
392 }

397 if (expected[0] == nullptr) {

415 bool readReply(char *reply, int maxLen) {
416 // skip empty lines
417 do {
418 if (!readReplyLine(reply, maxLen)) return false;
419 } while (strlen(reply) < 3);
420
421 // multi-line reply "123-..." ends with a line "123 ..."
422 if (reply[3] == '-') {
423 char line[FTP_RESULT_BUFFER_SIZE];
424 do {
425 if (!readReplyLine(line, sizeof(line))) return false;
426 FTPLogger.writeLog(LOG_DEBUG, "FTPBasicAPI::readReply", line);
427 } while (!(strncmp(line, reply, 3) == 0 && line[3] == ' '));
428 }
429 return true;
430 }

434 bool readReplyLine(char *str, int maxLen) {
435 int len = 0;
436 unsigned long start = millis();
437 while (true) {
438 if (command_ptr->available() > 0) {
439 int c = command_ptr->read();
440 if (c < 0) continue;
441 if (c == '\n') break;
442 if (len < maxLen - 1) str[len++] = (char)c;
443 } else if (!command_ptr->connected() ||
444 millis() - start >= FTP_REPLY_TIMEOUT_MS) {
445 str[len] = '\0';
446 FTPLogger.writeLog(LOG_ERROR, "FTPBasicAPI::readReplyLine",
447 "timeout or connection closed");
448 return false;
449 } else {
450 delay(1);
451 }
452 }
453 // For Windows we remove the \r at the end
454 if (len > 0 && str[len - 1] == '\r') len--;
455 str[len] = '\0';
456 return true;
457 }
